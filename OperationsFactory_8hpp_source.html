195template <typename T>
196template <typename OP>
197void sxeval::operations::OperationsFactory<T>::add() {
198 if constexpr (!(std::is_base_of<AOperation<T>, OP>::value)) {
199 throw std::invalid_argument("OP must be derived from AOperation");
200 }
201 if (_operations.find(OP::KEY) != _operations.end()) {
202 _operations.erase(OP::KEY);
203 }
204 const std::function<std::unique_ptr<AOperation<T>>(
205 const std::vector<AInstruction<T>*>&)> f =
206 [](const std::vector<AInstruction<T>*>& args) {
207 const auto nargs = static_cast<int>(args.size());
208 if (nargs < OP::ARITY_MIN) {
209 std::ostringstream oss;
210 oss << "Operation " << OP::KEY << " requires at least "
211 << OP::ARITY_MIN << " arguments";
212 throw std::invalid_argument(oss.str());
213 }
214 if (OP::ARITY_MAX != AOperation<T>::UNLIMITED_ARITY
215 && nargs > OP::ARITY_MAX)
216 {
217 std::ostringstream oss;
218 oss << "Operation " << OP::KEY << " requires at most "
219 << OP::ARITY_MAX << " arguments";
220 throw std::invalid_argument(oss.str());
221 }
222 return std::make_unique<OP>(args);
223 };
224 _operations.insert(std::make_pair(OP::KEY, f));
225}

227template <typename T>
228std::unique_ptr<sxeval::AOperation<T>>
229sxeval::operations::OperationsFactory<T>::create(
230 const std::string& key, const std::vector<sxeval::AInstruction<T>*> args)
231{
232 const auto it = _operations.find(key);
233 if (it == _operations.end()) {
234 throw std::invalid_argument("Unknown operation key: " + key);
235 }
236 return it->second(args);
237}

239template <typename T>
240T sxeval::operations::OperationsFactory<T>::compute(
241 const std::string& key, const std::vector<sxeval::AInstruction<T>*> args)
242 const
243{
244 const auto it = _operations.find(key);
245 if (it == _operations.end()) {
246 throw std::invalid_argument("Unknown operation key: " + key);
247 }
248 const auto op = it->second(args);
249 op->execute();
250 return op->getResult();
251}
